-- 1. Columna JSONB en usuario
ALTER TABLE usuario
    ADD COLUMN perfil_extendido JSONB;

ALTER TABLE usuario
    ADD CONSTRAINT ck_usuario_perfil_extendido_objeto
    CHECK (perfil_extendido IS NULL OR jsonb_typeof(perfil_extendido) = 'object');

-- 2. Otro dato semi-estructurado: preferencias de la publicacion / metadatos
ALTER TABLE publicacion
    ADD COLUMN metadatos JSONB;

ALTER TABLE publicacion
    ADD CONSTRAINT ck_publicacion_metadatos_objeto
    CHECK (metadatos IS NULL OR jsonb_typeof(metadatos) = 'object');

-- 3. Indices GIN para consultas eficientes sobre JSONB
CREATE INDEX idx_usuario_perfil_extendido ON usuario USING GIN (perfil_extendido);
CREATE INDEX idx_publicacion_metadatos    ON publicacion USING GIN (metadatos);

-- 7. Consultas
-- Buscar usuarios por tipo
SELECT u.id_usuario, u.nombres, u.apellidos, t.nombre AS tipo_usuario
FROM usuario u
JOIN tipo_usuario t ON t.id_tipo_usuario = u.id_tipo_usuario
WHERE t.nombre = 'estudiante';

-- Usuarios interesados en "IA"
SELECT nombres, perfil_extendido
FROM usuario
WHERE perfil_extendido -> 'intereses' ? 'IA';

-- Habilidades de cada usuario
SELECT nombres, perfil_extendido -> 'habilidades' AS habilidades
FROM usuario;

-- Docentes con linea de investigacion "Big Data"
SELECT nombres, perfil_extendido -> 'lineas_investigacion' AS lineas
FROM usuario
WHERE perfil_extendido @> '{"lineas_investigacion": ["Big Data"]}';

-- Publicaciones con la etiqueta "python"
SELECT id_publicacion, contenido, metadatos -> 'etiquetas' AS etiquetas
FROM publicacion
WHERE metadatos -> 'etiquetas' ? 'python';
